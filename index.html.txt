<script>
const $ = id => document.getElementById(id);

$("year").textContent = new Date().getFullYear();

$("countText").addEventListener("input", () => {
  const text = $("countText").value;
  const words = text.trim() ? text.trim().split(/\s+/u).length : 0;
  $("countResult").textContent =
    `الكلمات: ${words} | الحروف مع المسافات: ${text.length}`;
});

$("search").addEventListener("input", () => {
  const query = $("search").value.trim().toLowerCase();
  document.querySelectorAll(".card").forEach(card => {
    card.hidden = !card.dataset.name.toLowerCase().includes(query);
  });
});

function format(type) {
  let text = $("formatText").value;
  if (type === "clean") text = text.trim().replace(/\s+/gu, " ");
  if (type === "upper") text = text.toUpperCase();
  if (type === "lower") text = text.toLowerCase();
  $("formatResult").textContent = text;
}

function randomIndex(max) {
  const values = new Uint32Array(1);
  crypto.getRandomValues(values);
  return values[0] % max;
}

function randomString(chars, length) {
  let result = "";
  for (let i = 0; i < length; i++) {
    result += chars[randomIndex(chars.length)];
  }
  return result;
}

function generateWords() {
  const length = Number($("wordLength").value);
  const amount = Number($("wordAmount").value);
  const chars = "abcdefghijklmnopqrstuvwxyz0123456789_";
  const words = [];

  for (let i = 0; i < amount; i++) {
    words.push(randomString(chars, length));
  }

  $("wordResult").textContent = words.join("\n");
}

function generatePassword() {
  const length = Number($("passLength").value);
  const chars =
    "ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz23456789!@#$%&*?";
  $("passResult").textContent = randomString(chars, length);
}

async function copyResult(id) {
  const text = $(id).textContent.trim();
  if (!text) {
    alert("ولّد نتيجة أولًا!");
    return;
  }

  try {
    await navigator.clipboard.writeText(text);
    alert("تم نسخ النتيجة!");
  } catch {
    alert("تعذّر النسخ تلقائيًا. حدّد النتيجة وانسخها يدويًا.");
  }
}
